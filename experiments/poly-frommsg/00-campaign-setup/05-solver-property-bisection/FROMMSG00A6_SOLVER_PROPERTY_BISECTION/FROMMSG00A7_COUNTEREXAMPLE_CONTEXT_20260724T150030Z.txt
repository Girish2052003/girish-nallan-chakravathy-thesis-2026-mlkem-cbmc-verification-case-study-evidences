===== PRODUCTION FUNCTION — COMPRESS.C =====
   675	 *              The reference implementation contains the expression
   676	 *              `(msg[i] >> j) & 1` which the compiler can reason must
   677	 *              be either 0 or 1. */
   678	MLK_INTERNAL_API
   679	void mlk_poly_frommsg(mlk_poly *r, const uint8_t msg[MLKEM_INDCPA_MSGBYTES])
   680	{
   681	  unsigned i;
   682	#if (MLKEM_INDCPA_MSGBYTES != MLKEM_N / 8)
   683	#error "MLKEM_INDCPA_MSGBYTES must be equal to MLKEM_N/8 bytes!"
   684	#endif
   685	
   686	  for (i = 0; i < MLKEM_N / 8; i++)
   687	  __loop__(
   688	    invariant(i <= MLKEM_N / 8)
   689	    invariant(array_bound(r->coeffs, 0, 8 * i, 0, MLKEM_Q))
   690	    decreases(MLKEM_N / 8 - i))
   691	  {
   692	    unsigned j;
   693	    for (j = 0; j < 8; j++)
   694	    __loop__(
   695	      invariant(i <  MLKEM_N / 8 && j <= 8)
   696	      invariant(array_bound(r->coeffs, 0, 8 * i + j, 0, MLKEM_Q))
   697	      decreases(8 - j))
   698	    {
   699	      /* mlk_ct_sel_int16(MLKEM_Q_HALF, 0, b) is `Decompress_1(b != 0)`
   700	       * as per @[FIPS203, Eq (4.8)]. */
   701	
   702	      /* Prevent the compiler from recognizing this as a bit selection */
   703	      uint8_t mask = mlk_value_barrier_u8((uint8_t)(1u << j));
   704	      r->coeffs[8 * i + j] = mlk_ct_sel_int16(MLKEM_Q_HALF, 0, msg[i] & mask);
   705	    }
   706	  }
   707	  mlk_assert_abs_bound(r, MLKEM_N, MLKEM_Q);
   708	}
