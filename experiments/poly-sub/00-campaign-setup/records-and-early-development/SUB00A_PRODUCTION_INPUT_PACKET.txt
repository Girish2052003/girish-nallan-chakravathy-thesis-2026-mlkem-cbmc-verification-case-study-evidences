    31	/**
    32	 * Montgomery multiplication modulo MLKEM_Q.
    33	 *
    34	 * @reference{`fqmul()` in the reference implementation @[REF].}
    35	 *
    36	 * @param a First factor. Can be any int16_t.
    37	 * @param b Second factor. Must be signed canonical
    38	 *          (abs value < (MLKEM_Q+1)/2).
    39	 *
    40	 * @return 16-bit integer congruent to a*b*R^{-1} mod MLKEM_Q, and
    41	 *         smaller than MLKEM_Q in absolute value.
    42	 */
    43	static MLK_INLINE int16_t mlk_fqmul(int16_t a, int16_t b)
    44	__contract__(
    45	  requires(b > -MLKEM_Q_HALF && b < MLKEM_Q_HALF)
    46	  ensures(return_value > -MLKEM_Q && return_value < MLKEM_Q)
    47	)
    48	{
    49	  int16_t res;
    50	  mlk_assert_abs_bound(&b, 1, MLKEM_Q_HALF);
    51	
    52	  res = mlk_montgomery_reduce((int32_t)a * (int32_t)b);
    53	  /* Bounds:
    54	   * |res| <= ceil(|a| * |b| / 2^16) + (MLKEM_Q + 1) / 2
    55	   *       <= ceil(2^15 * ((MLKEM_Q - 1)/2) / 2^16) + (MLKEM_Q + 1) / 2
    56	   *       <= ceil((MLKEM_Q - 1) / 4) + (MLKEM_Q + 1) / 2
    57	   *        < MLKEM_Q
    58	   */
    59	
    60	  mlk_assert_abs_bound(&res, 1, MLKEM_Q);
    61	  return res;
    62	}

    64	/**
    65	 * Barrett reduction; given a 16-bit integer a, computes the centered
    66	 * representative congruent to a mod MLKEM_Q in [-(MLKEM_Q-1)/2, (MLKEM_Q-1)/2].
    67	 *
    68	 * @reference{`barrett_reduce()` in the reference implementation @[REF].}
    69	 *
    70	 * @param a Input integer to be reduced.
    71	 *
    72	 * @return Integer in [-(MLKEM_Q-1)/2, (MLKEM_Q-1)/2] congruent to @p a modulo
    73	 *         MLKEM_Q.
    74	 */
    75	static MLK_INLINE int16_t mlk_barrett_reduce(int16_t a)
    76	__contract__(
    77	  ensures(return_value > -MLKEM_Q_HALF && return_value < MLKEM_Q_HALF)
    78	)
    79	{
    80	  /* Barrett reduction approximates
    81	   * ```
    82	   *     round(a/MLKEM_Q)
    83	   *   = round(a*(2^N/MLKEM_Q))/2^N)
    84	   *  ~= round(a*round(2^N/MLKEM_Q)/2^N)
    85	   * ```
    86	   * Here, we pick N=26.
    87	   */
    88	  const int32_t magic = 20159; /* check-magic: 20159 == round(2^26 / MLKEM_Q) */
    89	
    90	  /*
    91	   * PORTABILITY: Right-shift on a signed integer is
    92	   * implementation-defined for negative left argument.
    93	   * Here, we assume it's sign-preserving "arithmetic" shift right.
    94	   * See (C99 6.5.7 (5))
    95	   */
    96	  const int32_t t = (magic * a + ((int32_t)1 << 25)) >> 26;
    97	
    98	  /*
    99	   * t is in -10 .. +10, so we need 32-bit math to
   100	   * evaluate t * MLKEM_Q and the subsequent subtraction
   101	   */
   102	  int16_t res = (int16_t)(a - t * MLKEM_Q);
   103	
   104	  mlk_assert_abs_bound(&res, 1, MLKEM_Q_HALF);
   105	  return res;
   106	}

   146	/**
   147	 * Constant-time conversion of signed representatives modulo MLKEM_Q within
   148	 * range [-(MLKEM_Q-1), MLKEM_Q-1] into unsigned representatives within
   149	 * range [0, MLKEM_Q-1].
   150	 *
   151	 * @reference{Not present in the reference implementation @[REF]. Used here
   152	 * to implement different semantics of `poly_reduce()`; see below. In the
   153	 * reference implementation @[REF] this logic is part of all compression
   154	 * functions (see `compress.c`).}
   155	 *
   156	 * @param c Signed coefficient to be converted.
   157	 *
   158	 * @return Unsigned representative in [0, MLKEM_Q).
   159	 */
   160	static MLK_INLINE int16_t mlk_scalar_signed_to_unsigned_q(int16_t c)
   161	__contract__(
   162	  requires(c > -MLKEM_Q && c < MLKEM_Q)
   163	  ensures(return_value >= 0 && return_value < MLKEM_Q)
   164	  ensures(return_value == (int32_t)c + (((int32_t)c < 0) * MLKEM_Q)))
   165	{
   166	  mlk_assert_abs_bound(&c, 1, MLKEM_Q);
   167	
   168	  /* Add MLKEM_Q if c is negative, but in constant time.
   169	   *
   170	   * Note that c + MLKEM_Q does not overflow in int16_t,
   171	   * so the cast to uint16_t is safe. */
   172	  c = mlk_ct_sel_int16((int16_t)(c + MLKEM_Q), c, mlk_ct_cmask_neg_i16(c));
   173	
   174	  mlk_assert_bound(&c, 1, 0, MLKEM_Q);
   175	  return c;
   176	}

   178	/* Reference: `poly_reduce()` in the reference implementation @[REF]
   179	 *            - We use _unsigned_ canonical outputs, while the reference
   180	 *              implementation uses _signed_ canonical outputs.
   181	 *              Accordingly, we need a conditional addition of MLKEM_Q
   182	 *              here to go from signed to unsigned representatives.
   183	 *              This conditional addition is then dropped from all
   184	 *              polynomial compression functions instead (see `compress.c`). */
   185	MLK_STATIC_TESTABLE void mlk_poly_reduce_c(mlk_poly *r)
   186	__contract__(
   187	  requires(memory_no_alias(r, sizeof(mlk_poly)))
   188	  assigns(memory_slice(r, sizeof(mlk_poly)))
   189	  ensures(array_bound(r->coeffs, 0, MLKEM_N, 0, MLKEM_Q))
   190	)
   191	{
   192	  unsigned i;
   193	
   194	  for (i = 0; i < MLKEM_N; i++)
   195	  __loop__(
   196	    invariant(i <= MLKEM_N)
   197	    invariant(array_bound(r->coeffs, 0, i, 0, MLKEM_Q))
   198	    decreases(MLKEM_N - i))
   199	  {
   200	    /* Barrett reduction, giving signed canonical representative */
   201	    int16_t t = mlk_barrett_reduce(r->coeffs[i]);
   202	    /* Conditional addition to get unsigned canonical representative */
   203	    r->coeffs[i] = mlk_scalar_signed_to_unsigned_q(t);
   204	  }
   205	
   206	  mlk_assert_bound(r, MLKEM_N, 0, MLKEM_Q);
   207	}

   225	/* Reference: `poly_add()` in the reference implementation @[REF].
   226	 *            - We use destructive version (output=first input) to avoid
   227	 *              reasoning about aliasing in the CBMC specification */
   228	MLK_INTERNAL_API
   229	void mlk_poly_add(mlk_poly *r, const mlk_poly *b)
   230	{
   231	  unsigned i;
   232	  for (i = 0; i < MLKEM_N; i++)
   233	  __loop__(
   234	    invariant(i <= MLKEM_N)
   235	    invariant(forall(k0, i, MLKEM_N, r->coeffs[k0] == loop_entry(*r).coeffs[k0]))
   236	    invariant(forall(k1, 0, i, r->coeffs[k1] == loop_entry(*r).coeffs[k1] + b->coeffs[k1]))
   237	    decreases(MLKEM_N - i))
   238	  {
   239	    /* The preconditions imply that the addition stays within int16_t. */
   240	    r->coeffs[i] = (int16_t)(r->coeffs[i] + b->coeffs[i]);
   241	  }
   242	}

   315	/*
   316	 * Computes a block CT butterflies with a fixed twiddle factor,
   317	 * using Montgomery multiplication.
   318	 * Parameters:
   319	 * - r: Pointer to base of polynomial (_not_ the base of butterfly block)
   320	 * - root: Twiddle factor to use for the butterfly. This must be in
   321	 *         Montgomery form and signed canonical.
   322	 * - start: Offset to the beginning of the butterfly block
   323	 * - len: Index difference between coefficients subject to a butterfly
   324	 * - bound: Ghost variable describing coefficient bound: Prior to `start`,
   325	 *          coefficients must be bound by `bound + MLKEM_Q`. Post `start`,
   326	 *          they must be bound by `bound`.
   327	 * When this function returns, output coefficients in the index range
   328	 * [start, start+2*len) have bound bumped to `bound + MLKEM_Q`.
   329	 * Example:
   330	 * - start=8, len=4
   331	 *   This would compute the following four butterflies
   332	 *          8     --    12
   333	 *             9    --     13
   334	 *                10   --     14
   335	 *                   11   --     15
   336	 * - start=4, len=2
   337	 *   This would compute the following two butterflies
   338	 *          4 -- 6
   339	 *             5 -- 7
   340	 */
   341	
   342	/* Reference: Embedded in `ntt()` in the reference implementation @[REF]. */
   343	static void mlk_ntt_butterfly_block(int16_t r[MLKEM_N], int16_t zeta,
   344	                                    unsigned start, unsigned len,
   345	                                    unsigned bound)
   346	__contract__(
   347	  requires(start < MLKEM_N)
   348	  requires(1 <= len && len <= MLKEM_N / 2 && start + 2 * len <= MLKEM_N)
   349	  requires(0 <= bound && bound < INT16_MAX - MLKEM_Q)
   350	  requires(-MLKEM_Q_HALF < zeta && zeta < MLKEM_Q_HALF)
   351	  requires(memory_no_alias(r, sizeof(int16_t) * MLKEM_N))
   352	  requires(array_abs_bound(r, 0, start, bound + MLKEM_Q))
   353	  requires(array_abs_bound(r, start, MLKEM_N, bound))
   354	  assigns(memory_slice(r, sizeof(int16_t) * MLKEM_N))
   355	  ensures(array_abs_bound(r, 0, start + 2*len, bound + MLKEM_Q))
   356	  ensures(array_abs_bound(r, start + 2 * len, MLKEM_N, bound)))
   357	{
   358	  /* `bound` is a ghost variable only needed in the CBMC specification */
   359	  unsigned j;
   360	  ((void)bound);
   361	  for (j = start; j < start + len; j++)
   362	  __loop__(
   363	    invariant(start <= j && j <= start + len)
   364	    /*
   365	     * Coefficients are updated in strided pairs, so the bounds for the
   366	     * intermediate states alternate twice between the old and new bound
   367	     */
   368	    invariant(array_abs_bound(r, 0,           j,           bound + MLKEM_Q))
   369	    invariant(array_abs_bound(r, j,           start + len, bound))
   370	    invariant(array_abs_bound(r, start + len, j + len,     bound + MLKEM_Q))
   371	    invariant(array_abs_bound(r, j + len,     MLKEM_N,     bound))
   372	    decreases(start + len - j))
   373	  {
   374	    int16_t t;
   375	    t = mlk_fqmul(r[j + len], zeta);
   376	    /* The precondition implies that the arithmetic does not overflow. */
   377	    r[j + len] = (int16_t)(r[j] - t);
   378	    r[j] = (int16_t)(r[j] + t);
   379	  }
   380	}

   382	/*
   383	 * Compute one layer of forward NTT
   384	 * Parameters:
   385	 * - r: Pointer to base of polynomial
   386	 * - layer: Variable indicating which layer is being applied.
   387	 */
   388	
   389	/* Reference: Embedded in `ntt()` in the reference implementation @[REF]. */
   390	static void mlk_ntt_layer(int16_t r[MLKEM_N], unsigned layer)
   391	__contract__(
   392	  requires(memory_no_alias(r, sizeof(int16_t) * MLKEM_N))
   393	  requires(1 <= layer && layer <= 7)
   394	  requires(array_abs_bound(r, 0, MLKEM_N, layer * MLKEM_Q))
   395	  assigns(memory_slice(r, sizeof(int16_t) * MLKEM_N))
   396	  ensures(array_abs_bound(r, 0, MLKEM_N, (layer + 1) * MLKEM_Q)))
   397	{
   398	  unsigned start, k, len;
   399	  /* Twiddle factors for layer n are at indices 2^(n-1)..2^n-1. */
   400	  k = 1u << (layer - 1);
   401	  len = (unsigned)MLKEM_N >> layer;
   402	  for (start = 0; start < MLKEM_N; start += 2 * len)
   403	  __loop__(
   404	    invariant(start < MLKEM_N + 2 * len)
   405	    invariant(k <= MLKEM_N / 2 && 2 * len * k == start + MLKEM_N)
   406	    invariant(array_abs_bound(r, 0, start, layer * MLKEM_Q + MLKEM_Q))
   407	    invariant(array_abs_bound(r, start, MLKEM_N, layer * MLKEM_Q))
   408	    decreases(MLKEM_N - start))
   409	  {
   410	    int16_t zeta = mlk_zetas[k++];
   411	    mlk_ntt_butterfly_block(r, zeta, start, len, layer * MLKEM_Q);
   412	  }
   413	}

   416	 * Compute full forward NTT
   417	 * NOTE: This particular implementation satisfies a much tighter
   418	 * bound on the output coefficients (5*q) than the contractual one (8*q),
   419	 * but this is not needed in the calling code. Should we change the
   420	 * base multiplication strategy to require smaller NTT output bounds,

   423	
   424	/* Reference: `ntt()` in the reference implementation @[REF].
   425	 * - Iterate over `layer` instead of `len` in the outer loop
   426	 *   to simplify computation of zeta index. */
   427	MLK_STATIC_TESTABLE void mlk_poly_ntt_c(mlk_poly *p)
   428	__contract__(
   429	  requires(memory_no_alias(p, sizeof(mlk_poly)))
   430	  requires(array_abs_bound(p->coeffs, 0, MLKEM_N, MLKEM_Q))
   431	  assigns(memory_slice(p, sizeof(mlk_poly)))
   432	  ensures(array_abs_bound(p->coeffs, 0, MLKEM_N, MLK_NTT_BOUND))
   433	)
   434	{
   435	  unsigned layer;
   436	  int16_t *r;
   437	
   438	  mlk_assert_abs_bound(p, MLKEM_N, MLKEM_Q);
   439	
   440	  r = p->coeffs;
   441	
   442	  for (layer = 1; layer <= 7; layer++)
   443	  __loop__(
   444	    invariant(1 <= layer && layer <= 8)
   445	    invariant(array_abs_bound(r, 0, MLKEM_N, layer * MLKEM_Q))
   446	    decreases(8 - layer))
   447	  {
   448	    mlk_ntt_layer(r, layer);
   449	  }
   450	
   451	  /* Check the stronger bound */

   473	/* Compute one layer of inverse NTT */
   474	
   475	/* Reference: Embedded into `invntt()` in the reference implementation @[REF] */
   476	static void mlk_invntt_layer(int16_t *r, unsigned layer)
   477	__contract__(
   478	  requires(memory_no_alias(r, sizeof(int16_t) * MLKEM_N))
   479	  requires(1 <= layer && layer <= 7)
   480	  requires(array_abs_bound(r, 0, MLKEM_N, MLKEM_Q))
   481	  assigns(memory_slice(r, sizeof(int16_t) * MLKEM_N))
   482	  ensures(array_abs_bound(r, 0, MLKEM_N, MLKEM_Q)))
   483	{
   484	  unsigned start, k, len;
   485	  len = (unsigned)MLKEM_N >> layer;
   486	  k = (1u << layer) - 1;
   487	  for (start = 0; start < MLKEM_N; start += 2 * len)
   488	  __loop__(
   489	    invariant(array_abs_bound(r, 0, MLKEM_N, MLKEM_Q))
   490	    invariant(start <= MLKEM_N && k <= 127)
   491	    /* Normalised form of k == MLKEM_N / len - 1 - start / (2 * len) */
   492	    invariant(2 * len * k + start == 2 * MLKEM_N - 2 * len)
   493	    decreases(MLKEM_N - start))
   494	  {
   495	    unsigned j;
   496	    int16_t zeta = mlk_zetas[k--];
   497	    for (j = start; j < start + len; j++)
   498	    __loop__(
   499	      invariant(start <= j && j <= start + len)
   500	      invariant(start <= MLKEM_N && k <= 127)
   501	      invariant(array_abs_bound(r, 0, MLKEM_N, MLKEM_Q))
   502	      decreases(start + len - j))
   503	    {
   504	      int16_t t = r[j];
   505	      /* The preconditions imply that the arithmetic does not overflow. */
   506	      r[j] = mlk_barrett_reduce((int16_t)(t + r[j + len]));
   507	      r[j + len] = (int16_t)(r[j + len] - t);
   508	      r[j + len] = mlk_fqmul(r[j + len], zeta);
   509	    }
   510	  }
   511	}
   512	
   513	/* Reference: `invntt()` in the reference implementation @[REF]
   514	 *            - We normalize at the beginning of the inverse NTT,
   515	 *              while the reference implementation normalizes at
   516	 *              the end. This allows us to drop a call to `poly_reduce()`
   517	 *              from the base multiplication. */
   518	MLK_STATIC_TESTABLE void mlk_poly_invntt_tomont_c(mlk_poly *p)
   519	__contract__(
   520	  requires(memory_no_alias(p, sizeof(mlk_poly)))
   521	  assigns(memory_slice(p, sizeof(mlk_poly)))
   522	  ensures(array_abs_bound(p->coeffs, 0, MLKEM_N, MLK_INVNTT_BOUND))
   523	)
   524	{
   525	  unsigned j, layer;
   526	  const int16_t f = 1441; /* check-magic: 1441 == pow(2,32 - 7,MLKEM_Q) */
   527	  int16_t *r = p->coeffs;
   528	
   529	  /*
   530	   * Scale input polynomial to account for Montgomery factor
   531	   * and NTT twist. This also brings coefficients down to
   532	   * absolute value < MLKEM_Q.
   533	   */
   534	  for (j = 0; j < MLKEM_N; j++)
   535	  __loop__(
   536	    invariant(j <= MLKEM_N)
   537	    invariant(array_abs_bound(r, 0, j, MLKEM_Q))
   538	    decreases(MLKEM_N - j))
   539	  {
   540	    r[j] = mlk_fqmul(r[j], f);
   541	  }
   542	
   543	  /* Run the invNTT layers */
   544	  for (layer = 7; layer > 0; layer--)
   545	  __loop__(
   546	    invariant(0 <= layer && layer < 8)
   547	    invariant(array_abs_bound(r, 0, MLKEM_N, MLKEM_Q))
   548	    decreases(layer))
   549	  {
   550	    mlk_invntt_layer(r, layer);
   551	  }
   552	
   553	  mlk_assert_abs_bound(p, MLKEM_N, MLK_INVNTT_BOUND);
   554	}

   565	    return;
   566	  }
   567	#endif /* MLK_USE_NATIVE_INTT */
   568	
   569	  mlk_poly_invntt_tomont_c(r);

    48	/**
    49	 * Generic Montgomery reduction; given a 32-bit integer a, computes a 16-bit
    50	 * integer congruent to a * R^-1 mod MLKEM_Q, where R=2^16.
    51	 *
    52	 * @param a Input integer to be reduced, of absolute value smaller or equal
    53	 *          to INT32_MAX - 2^15 * MLKEM_Q.
    54	 *
    55	 * @return Integer congruent to a * R^-1 modulo MLKEM_Q, with absolute value
    56	 *         <= ceil(|a| / 2^16) + (MLKEM_Q + 1)/2.
    57	 */
    58	static MLK_ALWAYS_INLINE int16_t mlk_montgomery_reduce(int32_t a)
    59	__contract__(
    60	    requires(a < +(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)) &&
    61	             a > -(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)))
    62	    /* We don't attempt to express an input-dependent output bound
    63	     * as the post-condition here. There are two call-sites for this
    64	     * function:
    65	     * - The base multiplication: Here, we need no output bound.
    66	     * - mlk_fqmul: Here, we inline this function and prove another spec
    67	     *          for mlk_fqmul which does have a post-condition bound. */
    68	)
    69	{
    70	  /* check-magic: 62209 == unsigned_mod(pow(MLKEM_Q, -1, 2^16), 2^16) */
    71	  const uint32_t QINV = 62209;
    72	
    73	  /* Compute a*q^{-1} mod 2^16 in unsigned representatives. */
    74	  const uint16_t a_reduced = mlk_cast_int32_to_uint16(a);
    75	  const uint16_t a_inverted = (a_reduced * QINV) & UINT16_MAX;
    76	
    77	  /* Lift to signed canonical representative mod 2^16. */
    78	  const int16_t t = mlk_cast_uint16_to_int16(a_inverted);
    79	
    80	  int32_t r;
    81	
    82	  mlk_assert(a < +(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)) &&
    83	             a > -(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)));
    84	
    85	  r = a - ((int32_t)t * MLKEM_Q);
    86	
    87	  /*
    88	   * PORTABILITY: Right-shift on a signed integer is, strictly-speaking,
    89	   * implementation-defined for negative left argument. Here,
    90	   * we assume it's sign-preserving "arithmetic" shift right. (C99 6.5.7 (5))
    91	   */
    92	  r = r >> 16;
    93	  /* Bounds: |r >> 16| <= ceil(|r| / 2^16)
    94	   *                   <= ceil(|a| / 2^16 + MLKEM_Q / 2)
    95	   *                   <= ceil(|a| / 2^16) + (MLKEM_Q + 1) / 2
    96	   *
    97	   * (Note that |a >> n| = ceil(|a| / 2^16) for negative a)
    98	   */
    99	  return (int16_t)r;
   100	}

    37	/* Constant-time comparisons and conditional operations
    38	
    39	   We reduce the risk for compilation into variable-time code
    40	   through the use of 'value barriers'.
    41	
    42	   Functionally, a value barrier is a no-op. To the compiler, however,
    43	   it constitutes an arbitrary modification of its input, and therefore
    44	   harden's value propagation and range analysis.
    45	
    46	   We consider two approaches to implement a value barrier:
    47	   - An empty inline asm block which marks the target value as clobbered.
    48	   - XOR'ing with the value of a volatile global that's set to 0;
    49	     see @[optblocker] for a discussion of this idea, and
    50	     @[libmceliece, inttypes/crypto_intN.h] for an implementation.
    51	
    52	   The first approach is cheap because it only prevents the compiler
    53	   from reasoning about the value of the variable past the barrier,
    54	   but does not directly generate additional instructions.
    55	
    56	   The second approach generates redundant loads and XOR operations
    57	   and therefore comes at a higher runtime cost. However, it appears
    58	   more robust towards optimization, as compilers should never drop
    59	   a volatile load.
    60	
    61	   We use the empty-ASM value barrier for GCC and clang, and fall
    62	   back to the global volatile barrier otherwise.
    63	
    64	   The global value barrier can be forced by setting
    65	   MLK_CONFIG_NO_ASM_VALUE_BARRIER.
    66	
    67	*/

   202	 * @return Mask value (0 or 0xFFFF).
   203	 */
   204	static MLK_INLINE uint16_t mlk_ct_cmask_neg_i16(int16_t x)
   205	__contract__(ensures(return_value == ((x < 0) ? 0xFFFF : 0)))
   206	{
   207	  int32_t tmp = mlk_value_barrier_i32((int32_t)x);
   208	  /*
   209	   * PORTABILITY: Right-shift on a signed integer is
   210	   * implementation-defined for negative left argument.
   211	   * Here, we assume it's sign-preserving "arithmetic" shift right.
   212	   * See (C99 6.5.7 (5))
   213	   */
   214	  tmp >>= 16;
   215	  return mlk_cast_int32_to_uint16(tmp);
   216	}
   217	
   218	/**
   219	 * Return 0 if input is zero, and -1 otherwise.
   220	 *
   221	 * @reference{Embedded in `cmov_int16()` in the reference implementation
   222	 * @[REF]. Uses a value barrier and shift instead of `b = -b` to convert
   223	 * condition into mask.}
   224	 *
   225	 * @param x Value to be converted into a mask.
   226	 *
   227	 * @return Mask value (0 or 0xFFFF).
   228	 */
   229	static MLK_INLINE uint16_t mlk_ct_cmask_nonzero_u16(uint16_t x)
   230	__contract__(ensures(return_value == ((x == 0) ? 0 : 0xFFFF)))
   231	{
   232	  int32_t tmp = mlk_value_barrier_i32(-((int32_t)x));
   233	  /*
   234	   * PORTABILITY: Right-shift on a signed integer is
   235	   * implementation-defined for negative left argument.
   236	   * Here, we assume it's sign-preserving "arithmetic" shift right.
   237	   * See (C99 6.5.7 (5))
   238	   */
   239	  tmp >>= 16;
   240	  return mlk_cast_int32_to_uint16(tmp);
   241	}
   242	
   243	/**
   244	 * Return 0 if input is zero, and -1 otherwise.
   245	 *
   246	 * @reference{Embedded in `verify()` and `cmov()` in the reference
   247	 * implementation @[REF]. We include a value barrier not present in the
   248	 * reference implementation, to prevent the compiler from realizing that
   249	 * this function returns a mask.}
   250	 *
   251	 * @param x Value to be converted into a mask.
   252	 *
   253	 * @return Mask value (0 or 0xFF).
   254	 */
   255	static MLK_INLINE uint8_t mlk_ct_cmask_nonzero_u8(uint8_t x)
   256	__contract__(ensures(return_value == ((x == 0) ? 0 : 0xFF)))
   257	{
   258	  uint16_t mask = mlk_ct_cmask_nonzero_u16((uint16_t)x);
   259	  return (uint8_t)(mask & 0xFF);
   260	}
   261	
   262	/**
   263	 * Functionally equivalent to cond ? a : b, but implemented with guards
   264	 * against compiler-introduced branches.
   265	 *
   266	 * @spec{With `a = MLKEM_Q_HALF` and `b=0`, this essentially implements
   267	 * `Decompress_1` @[FIPS203, Eq (4.8)] in `mlk_poly_frommsg()`. With
   268	 * `a = x + MLKEM_Q`, `b = x`, and `cond` indicating whether `x` is negative,
   269	 * implements signed->unsigned conversion of modular representatives.
   270	 * Questions of representation are not considered in the specification
   271	 * @[FIPS203, Section 2.4.1, "The pseudocode is agnostic regarding how an
   272	 * integer modulo 𝑚 is represented in actual implementations"].}
   273	 *
   274	 * @reference{Embedded in the polynomial compression function in the
   275	 * reference implementation @[REF]. Used as part of signed->unsigned
   276	 * conversion for modular representatives. This happens in `mlk_poly_reduce()`
   277	 * here, and as part of polynomial compression functions in @[REF]. See
   278	 * `mlk_poly_reduce()`. Barrier to reduce the risk of compiler-introduced
   279	 * branches. For `a = MLKEM_Q_HALF` and `b=0`, also embedded in
   280	 * `poly_frommsg()` from the reference implementation, which uses
   281	 * `cmov_int16()` instead.}
   282	 *
   283	 * @param a    First alternative.
   284	 * @param b    Second alternative.
   285	 * @param cond Condition variable.
   286	 *
   287	 * @return @p a if @p cond != 0, else @p b.
   288	 */
   289	static MLK_INLINE int16_t mlk_ct_sel_int16(int16_t a, int16_t b, uint16_t cond)
   290	__contract__(ensures(return_value == (cond ? a : b)))
   291	{
   292	  uint16_t au = mlk_cast_int16_to_uint16(a);
   293	  uint16_t bu = mlk_cast_int16_to_uint16(b);
   294	  uint16_t res = bu ^ (mlk_ct_cmask_nonzero_u16(cond) & (au ^ bu));
   295	  return mlk_cast_uint16_to_int16(res);
   296	}
   297	
   298	/**
   299	 * Functionally equivalent to cond ? a : b, but implemented with guards
   300	 * against compiler-introduced branches.
   301	 *
   302	 * @reference{Embedded into `cmov()` in the reference implementation @[REF].
   303	 * Uses a value barrier to get mask from condition value.}
   304	 *
   305	 * @param a    First alternative.
   306	 * @param b    Second alternative.
   307	 * @param cond Condition variable.
   308	 *
   309	 * @return @p a if @p cond != 0, else @p b.
   310	 */
   311	static MLK_INLINE uint8_t mlk_ct_sel_uint8(uint8_t a, uint8_t b, uint8_t cond)
   312	__contract__(ensures(return_value == (cond ? a : b)))
   313	{
   314	  return b ^ (mlk_ct_cmask_nonzero_u8(cond) & (a ^ b));
   315	}
   316	
   317	/**
   318	 * Compare two arrays for equality in constant time.
   319	 *
   320	 * @spec{Used to securely compute conditional move in @[FIPS203, Algorithm
   321	 * 18 (ML-KEM.Decaps_Internal, L9-11].}
   322	 *
   323	 * @reference{`cmov()` in the reference implementation @[REF]. We return
   324	 * `uint8_t`, not `int`. We use an additional XOR-accumulator in the
   325	 * comparison loop which prevents early abort if the OR-accumulator is 0xFF.
   326	 * We use a value barrier to convert the OR-accumulator into a mask; the
   327	 * reference implementation uses a shift which the compiler can argue to
   328	 * result in either 0 or 0xFF..FF.}
   329	 *
   330	 * @param[in] a   First byte array.
   331	 * @param[in] b   Second byte array.
   332	 * @param     len Length of the byte arrays, upper-bounded to UINT16_MAX to
   333	 *                control proof complexity only.
   334	 *
   335	 * @retval 0    The byte arrays are equal.
   336	 * @retval 0xFF The byte arrays are not equal.
   337	 */
   338	static MLK_INLINE uint8_t mlk_ct_memcmp(const uint8_t *a, const uint8_t *b,
   339	                                        const size_t len)
   340	__contract__(
   341	  requires(len <= UINT16_MAX)
   342	  requires(memory_no_alias(a, len))
   343	  requires(memory_no_alias(b, len))
   344	  ensures((return_value == 0) || (return_value == 0xFF))
   345	  ensures((return_value == 0) == forall(i, 0, len, (a[i] == b[i]))))
   346	{
   347	  uint8_t r = 0, s = 0;
   348	  unsigned i;
   349	
   350	  for (i = 0; i < len; i++)
   351	  __loop__(
   352	    invariant(i <= len)
   353	    invariant((r == 0) == (forall(k, 0, i, (a[k] == b[k]))))
   354	    decreases(len - i))
   355	  {
   356	    r |= a[i] ^ b[i];
   357	    /* s is useless, but prevents the loop from being aborted once r=0xff. */
   358	    s ^= a[i] ^ b[i];
   359	  }
   360	
   361	  /*
   362	   * - Convert r into a mask; this may not be necessary, but is an additional
   363	   *   safeguard
   364	   *   towards leaking information about a and b.
   365	   * - XOR twice with s, separated by a value barrier, to prevent the compile
   366	   *   from dropping the s computation in the loop.
   367	   */
   368	  return (mlk_value_barrier_u8(mlk_ct_cmask_nonzero_u8(r) ^ s) ^ s);
   369	}
   370	
   371	/**
   372	 * Copy len bytes from x to r if b is zero; don't modify x if b is non-zero.
   373	 * Assumes two's complement representation of negative integers. Runs in
   374	 * constant time.
   375	 *
   376	 * @spec{Used to securely compute conditional move in @[FIPS203, Algorithm
   377	 * 18 (ML-KEM.Decaps_Internal, L9-11].}
   378	 *
   379	 * @reference{`cmov()` in the reference implementation @[REF]. We move if
   380	 * condition value is `0`, not `1`. We use `mlk_ct_sel_uint8` for
   381	 * constant-time selection.}
   382	 *
   383	 * @param[out] r   Output byte array.
   384	 * @param[in]  x   Input byte array.
   385	 * @param      len Number of bytes to be copied.
   386	 * @param      b   Condition value.
   387	 */
   388	static MLK_INLINE void mlk_ct_cmov_zero(uint8_t *r, const uint8_t *x,
   389	                                        size_t len, uint8_t b)
   390	__contract__(
   391	  requires(len <= UINT32_MAX)
   392	  requires(memory_no_alias(r, len))
   393	  requires(memory_no_alias(x, len))
   394	  assigns(memory_slice(r, len))
   395	  ensures(forall(i, 0, len, (r[i] == (b == 0 ? x[i] : old(r)[i])))))
   396	{
   397	  size_t i;
   398	  for (i = 0; i < len; i++)
   399	  __loop__(
   400	    invariant(i <= len)
   401	    invariant(forall(k, 0, i, r[k] == (b == 0 ? x[k] : loop_entry(r)[k])))
   402	    decreases(len - i))
   403	  {
   404	    r[i] = mlk_ct_sel_uint8(r[i], x[i], b);
   405	  }
   406	}
