     2	 * MSG-T2-R2B expected-failure mutation.
     3	 *
     4	 * Correct theorem antecedent:
     5	 *
     6	 *   A[i] == B[i] for every i != k.
     7	 *
     8	 * Coefficient k alone may differ.
     9	 *
    10	 * Deliberately corrupted antecedent:
    11	 *
    12	 *   coefficient k differs, and one additional coefficient m differs.
    13	 *
    14	 * The second differing coefficient m is selected from a different output
    15	 * byte. The corrupted relation nevertheless claims that output bit m remains
    16	 * equal. CBMC must reject that claim.
    17	 *
    18	 * This is functional mutation evidence, not a timing or leakage claim.
    19	 */
    20	
    21	#include <stdint.h>
    22	#include "compress.h"
    23	
    24	int main(void)
    25	{
    26	  mlk_poly a;
    27	  mlk_poly b;

    29	  uint8_t msg_a[MLKEM_INDCPA_MSGBYTES];
    30	  uint8_t msg_b[MLKEM_INDCPA_MSGBYTES];
    31	
    32	  unsigned i;
    33	  unsigned k;
    34	  unsigned m;
    35	
    36	  uint8_t bit_a_m;
    37	  uint8_t bit_b_m;
    38	
    39	  __CPROVER_assume(k < MLKEM_N);
    40	  __CPROVER_assume(m < MLKEM_N);
    41	  __CPROVER_assume(m != k);
    42	
    43	  /*
    44	   * Ensure m belongs to a different output byte from k. The mutation
    45	   * therefore directly attacks other-byte and all-other-bit preservation.
    46	   */
    47	  __CPROVER_assume((m >> 3) != (k >> 3));
    48	
    49	  for (i = 0u; i < MLKEM_N; i++)
    50	  {
    51	    __CPROVER_assume(a.coeffs[i] >= 0);
    52	    __CPROVER_assume(a.coeffs[i] < MLKEM_Q);
    53	
    54	    __CPROVER_assume(b.coeffs[i] >= 0);
    55	    __CPROVER_assume(b.coeffs[i] < MLKEM_Q);
    56	
    57	    if (i != k && i != m)
    58	    {
    59	      __CPROVER_assume(a.coeffs[i] == b.coeffs[i]);
    60	    }
    61	  }
    62	
    63	  /*
    64	   * The selected coefficient k is the difference permitted by R2B.
    65	   */
    66	  __CPROVER_assume(a.coeffs[k] != b.coeffs[k]);
    67	
    68	  /*
    69	   * Deliberate second difference:
    70	   *

    73	   */
    74	  __CPROVER_assume(a.coeffs[m] == 0);
