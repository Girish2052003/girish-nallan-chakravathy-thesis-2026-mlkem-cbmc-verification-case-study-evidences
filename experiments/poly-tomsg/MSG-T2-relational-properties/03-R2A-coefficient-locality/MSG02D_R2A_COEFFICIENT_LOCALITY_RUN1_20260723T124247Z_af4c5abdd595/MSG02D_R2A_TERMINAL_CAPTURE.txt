     1	/*
     2	 * MSG-T2-R2A — symbolic coefficient locality.
     3	 *
     4	 * For arbitrary canonical polynomials A and B and arbitrary k:
     5	 *
     6	 *   A[k] == B[k]
     7	 *
     8	 * implies:
     9	 *
    10	 *   bit_k(tomsg(A)) == bit_k(tomsg(B)).
    11	 *
    12	 * All coefficients other than k remain mutually unrestricted. Therefore the
    13	 * output bit at position k is proved to depend only on coefficient k, not on
    14	 * any other polynomial coefficient.
    15	 *
    16	 * This is functional non-interference/locality. It is not a timing,
    17	 * microarchitectural, leakage, or constant-time claim.
    18	 */
    19	
    20	#include <stdint.h>
    21	#include "compress.h"
    22	
    23	int main(void)
    24	{
    25	  mlk_poly a;
    26	  mlk_poly b;
    27	
    28	  uint8_t msg_a[MLKEM_INDCPA_MSGBYTES];
    29	  uint8_t msg_b[MLKEM_INDCPA_MSGBYTES];
    30	
    31	  unsigned i;
    32	  unsigned k;
    33	
    34	  uint8_t bit_a;
    35	  uint8_t bit_b;

    37	  __CPROVER_assert(
    38	      MLKEM_N == 256,
    39	      "MSG_T2_R2A_MODEL: polynomial degree must be 256");
    40	
    41	  __CPROVER_assert(
    42	      MLKEM_INDCPA_MSGBYTES == 32,
    43	      "MSG_T2_R2A_MODEL: message size must be 32 bytes");
    44	
    45	  /*
    46	   * Both input polynomials range independently over the complete canonical
    47	   * domain.
    48	   */

    50	  {
    51	    __CPROVER_assume(a.coeffs[i] >= 0);

    53

    55	    __CPROVER_assume(b.coeffs[i] < MLKEM_Q);
    56	  }
    57	
    58	  /*
    59	   * Select an arbitrary coefficient position and equate only that
    60	   * coefficient. All other A/B coefficient pairs remain unrestricted.
    61	   */
    62	  __CPROVER_assume(k < MLKEM_N);
    63	  __CPROVER_assume(a.coeffs[k] == b.coeffs[k]);
