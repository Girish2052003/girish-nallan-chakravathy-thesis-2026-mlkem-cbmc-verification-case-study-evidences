     2	 * MSG-T2-R2A expected-failure mutation.
     3	 *
     4	 * Correct theorem:
     5	 *
     6	 *   A[k] == B[k]
     7	 *     implies
     8	 *   bit_k(tomsg(A)) == bit_k(tomsg(B)).
     9	 *
    10	 * Deliberately corrupted theorem:
    11	 *
    12	 *   A[k] != B[k]
    13	 *     but still assert
    14	 *   bit_k(tomsg(A)) == bit_k(tomsg(B)).
    15	 *
    16	 * Every coefficient other than k is held equal. Therefore a counterexample
    17	 * is isolated to the selected coefficient and cannot be attributed to an
    18	 * unrelated polynomial coefficient.
    19	 */
    20	
    21	#include <stdint.h>

    36	  uint8_t bit_b;

    39

    56	   * rather than equal.
    57	   */
