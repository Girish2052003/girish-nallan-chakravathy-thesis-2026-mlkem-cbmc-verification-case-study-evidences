     1	/*
     2	 * MSG-T2-R3B — determinism and input-frame preservation.
     3	 *
     4	 * Let SOURCE be an arbitrary canonical polynomial. Construct two independent
     5	 * polynomial objects RUN_A and RUN_B with identical initial contents:
     6	 *
     7	 *   RUN_A_before == RUN_B_before == SOURCE.
     8	 *
     9	 * Execute the real production function once on each object, using separate
    10	 * destination buffers whose initial contents are not constrained equal.
    11	 *
    12	 * The theorem establishes:
    13	 *
    14	 *   1. RUN_A remains equal to SOURCE after execution;
    15	 *   2. RUN_B remains equal to SOURCE after execution;
    16	 *   3. every byte of MSG_A equals the corresponding byte of MSG_B.
    17	 *
    18	 * Thus equal polynomial values produce equal complete message values,
    19	 * independent of input-object identity and destination-buffer identity.
    20	 *
    21	 * This is functional determinism. It is not a timing, scheduling,
    22	 * microarchitectural, leakage, or constant-time claim.
    23	 */
    24	
    25	#include <stdint.h>
    26	#include "compress.h"
    27	
    28	int main(void)
    29	{
    30	  mlk_poly source;
    31	  mlk_poly run_a;
    32	  mlk_poly run_b;
    33	
    34	  uint8_t msg_a[MLKEM_INDCPA_MSGBYTES];
    35	  uint8_t msg_b[MLKEM_INDCPA_MSGBYTES];
    36

    38	  unsigned byte_index;

    49	   * SOURCE ranges over the complete canonical polynomial domain.
    50	   */
    51	  for (i = 0u; i < MLKEM_N; i++)

    59	   */
