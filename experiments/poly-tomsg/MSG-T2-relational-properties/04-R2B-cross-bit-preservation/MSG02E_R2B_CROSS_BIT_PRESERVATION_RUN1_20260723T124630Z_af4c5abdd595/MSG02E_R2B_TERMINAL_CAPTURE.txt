     2	 * MSG-T2-R2B — symbolic cross-bit preservation.
     3	 *
     4	 * Let A and B be arbitrary canonical polynomials and let k be an arbitrary
     5	 * coefficient index.
     6	 *
     7	 * Assume:
     8	 *
     9	 *   A[i] == B[i] for every i != k.
    10	 *
    11	 * Coefficient k remains unrestricted and may either remain equal or change.
    12	 *
    13	 * The theorem establishes that changing at most coefficient k cannot change:
    14	 *
    15	 *   1. any output byte outside byte floor(k / 8);
    16	 *   2. any of the other seven bits inside byte floor(k / 8);
    17	 *   3. equivalently, any output bit j where j != k.
    18	 *
    19	 * This is functional cross-bit non-interference. It is not a timing,
    20	 * microarchitectural, leakage, or constant-time claim.
    21	 */

    24	#include "compress.h"

    32	  uint8_t msg_b[MLKEM_INDCPA_MSGBYTES];

    35	  unsigned j;

    37	  unsigned byte_index;
