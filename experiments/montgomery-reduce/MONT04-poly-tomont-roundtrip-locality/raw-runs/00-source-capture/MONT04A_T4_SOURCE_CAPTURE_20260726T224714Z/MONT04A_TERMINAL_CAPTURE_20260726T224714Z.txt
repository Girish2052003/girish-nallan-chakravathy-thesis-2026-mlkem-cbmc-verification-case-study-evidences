    48	/**
    49	 * Generic Montgomery reduction; given a 32-bit integer a, computes a 16-bit
    50	 * integer congruent to a * R^-1 mod MLKEM_Q, where R=2^16.
    51	 *
    52	 * @param a Input integer to be reduced, of absolute value smaller or equal
    53	 *          to INT32_MAX - 2^15 * MLKEM_Q.
    54	 *
    55	 * @return Integer congruent to a * R^-1 modulo MLKEM_Q, with absolute value
    56	 *         <= ceil(|a| / 2^16) + (MLKEM_Q + 1)/2.
    57	 */
    58	static MLK_ALWAYS_INLINE int16_t mlk_montgomery_reduce(int32_t a)
    59	__contract__(
    60	    requires(a < +(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)) &&
    61	             a > -(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)))
    62	    /* We don't attempt to express an input-dependent output bound
    63	     * as the post-condition here. There are two call-sites for this
    64	     * function:
    65	     * - The base multiplication: Here, we need no output bound.
    66	     * - mlk_fqmul: Here, we inline this function and prove another spec
    67	     *          for mlk_fqmul which does have a post-condition bound. */
    68	)
    69	{
    70	  /* check-magic: 62209 == unsigned_mod(pow(MLKEM_Q, -1, 2^16), 2^16) */
    71	  const uint32_t QINV = 62209;
    72	
    73	  /* Compute a*q^{-1} mod 2^16 in unsigned representatives. */
    74	  const uint16_t a_reduced = mlk_cast_int32_to_uint16(a);
    75	  const uint16_t a_inverted = (a_reduced * QINV) & UINT16_MAX;
    76	
    77	  /* Lift to signed canonical representative mod 2^16. */
    78	  const int16_t t = mlk_cast_uint16_to_int16(a_inverted);
    79	
    80	  int32_t r;
    81	
    82	  mlk_assert(a < +(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)) &&
    83	             a > -(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)));
    84	
    85	  r = a - ((int32_t)t * MLKEM_Q);
    86	
    87	  /*
    88	   * PORTABILITY: Right-shift on a signed integer is, strictly-speaking,
    89	   * implementation-defined for negative left argument. Here,
    90	   * we assume it's sign-preserving "arithmetic" shift right. (C99 6.5.7 (5))
    91	   */

    93	  /* Bounds: |r >> 16| <= ceil(|r| / 2^16)
    94	   *                   <= ceil(|a| / 2^16 + MLKEM_Q / 2)
    95	   *                   <= ceil(|a| / 2^16) + (MLKEM_Q + 1) / 2
    96	   *
    97	   * (Note that |a >> n| = ceil(|a| / 2^16) for negative a)
    98	   */
