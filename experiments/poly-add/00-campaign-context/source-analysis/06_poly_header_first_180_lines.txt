    30	/**

    48	/**
    49	 * Generic Montgomery reduction; given a 32-bit integer a, computes a 16-bit
    50	 * integer congruent to a * R^-1 mod MLKEM_Q, where R=2^16.
    51	 *
    52	 * @param a Input integer to be reduced, of absolute value smaller or equal
    53	 *          to INT32_MAX - 2^15 * MLKEM_Q.
    54	 *
    55	 * @return Integer congruent to a * R^-1 modulo MLKEM_Q, with absolute value
    56	 *         <= ceil(|a| / 2^16) + (MLKEM_Q + 1)/2.
    57	 */
    58	static MLK_ALWAYS_INLINE int16_t mlk_montgomery_reduce(int32_t a)
    59	__contract__(
    60	    requires(a < +(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)) &&
    61	             a > -(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)))
    62	    /* We don't attempt to express an input-dependent output bound
    63	     * as the post-condition here. There are two call-sites for this
    64	     * function:
    65	     * - The base multiplication: Here, we need no output bound.
    66	     * - mlk_fqmul: Here, we inline this function and prove another spec
    67	     *          for mlk_fqmul which does have a post-condition bound. */
    68	)
    69	{
    70	  /* check-magic: 62209 == unsigned_mod(pow(MLKEM_Q, -1, 2^16), 2^16) */
    71	  const uint32_t QINV = 62209;
    72	
    73	  /* Compute a*q^{-1} mod 2^16 in unsigned representatives. */
    74	  const uint16_t a_reduced = mlk_cast_int32_to_uint16(a);
    75	  const uint16_t a_inverted = (a_reduced * QINV) & UINT16_MAX;
    76	
    77	  /* Lift to signed canonical representative mod 2^16. */
    78	  const int16_t t = mlk_cast_uint16_to_int16(a_inverted);
    79	
    80	  int32_t r;
    81	
    82	  mlk_assert(a < +(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)) &&
    83	             a > -(INT32_MAX - (((int32_t)1 << 15) * MLKEM_Q)));
    84	
    85	  r = a - ((int32_t)t * MLKEM_Q);
    86	
    87	  /*
    88	   * PORTABILITY: Right-shift on a signed integer is, strictly-speaking,
    89	   * implementation-defined for negative left argument. Here,
    90	   * we assume it's sign-preserving "arithmetic" shift right. (C99 6.5.7 (5))
    91	   */
    92	  r = r >> 16;
    93	  /* Bounds: |r >> 16| <= ceil(|r| / 2^16)
    94	   *                   <= ceil(|a| / 2^16 + MLKEM_Q / 2)
    95	   *                   <= ceil(|a| / 2^16) + (MLKEM_Q + 1) / 2
    96	   *
    97	   * (Note that |a >> n| = ceil(|a| / 2^16) for negative a)
    98	   */
    99	  return (int16_t)r;
   100	}

   102	#define mlk_poly_tomont MLK_NAMESPACE(poly_tomont)
   103	/**
   104	 * In-place conversion of all coefficients of a polynomial from the normal
   105	 * domain to the Montgomery domain.
   106	 *
   107	 * Bounds: output < MLKEM_Q in absolute value.
   108	 *
   109	 * @spec{Internal normalization required in `mlk_indcpa_keypair_derand` as
   110	 * part of matrix-vector multiplication @[FIPS203, Algorithm 13, K-PKE.KeyGen,
   111	 * L18].}
   112	 *
   113	 * @param[in,out] r Input/output polynomial.
   114	 */
   115	MLK_INTERNAL_API
   116	void mlk_poly_tomont(mlk_poly *r)
   117	__contract__(
   118	  requires(memory_no_alias(r, sizeof(mlk_poly)))
   119	  assigns(memory_slice(r, sizeof(mlk_poly)))
   120	  ensures(array_abs_bound(r->coeffs, 0, MLKEM_N, MLKEM_Q))
   121	);

   123	#define mlk_poly_mulcache_compute MLK_NAMESPACE(poly_mulcache_compute)
   124	/**
   125	 * Compute the mulcache for a polynomial in NTT domain.
   126	 *
   127	 * The mulcache of a degree-2 polynomial b := b0 + b1*X in Fq[X]/(X^2-zeta)
   128	 * is the value b1*zeta, needed when computing products of b in
   129	 * Fq[X]/(X^2-zeta).
   130	 *
   131	 * The mulcache of a polynomial in NTT domain -- which is a 128-tuple of
   132	 * degree-2 polynomials in Fq[X]/(X^2-zeta), for varying zeta, is the
   133	 * 128-tuple of mulcaches of those polynomials.
   134	 *
   135	 * @spec{Caches `b_1 * \gamma` in @[FIPS203, Algorithm 12, BaseCaseMultiply,
   136	 * L1].}
   137	 *
   138	 * @param[out] x Mulcache to be populated.
   139	 * @param[in]  a Input polynomial.
   140	 */
   141	/*
   142	 * NOTE: The default C implementation of this function populates
   143	 * the mulcache with values in (-q,q), but this is not needed for the
   144	 * higher level safety proofs, and thus not part of the spec.
   145	 */
   146	MLK_INTERNAL_API
   147	void mlk_poly_mulcache_compute(mlk_poly_mulcache *x, const mlk_poly *a)
   148	__contract__(
   149	  requires(memory_no_alias(x, sizeof(mlk_poly_mulcache)))
   150	  requires(memory_no_alias(a, sizeof(mlk_poly)))
   151	  assigns(memory_slice(x, sizeof(mlk_poly_mulcache)))
   152	);

   154	#define mlk_poly_reduce MLK_NAMESPACE(poly_reduce)
   155	/**
   156	 * Convert a polynomial to unsigned canonical representatives.
   157	 *
   158	 * The input coefficients can be arbitrary integers in int16_t. The output
   159	 * coefficients are in [0,1,..,MLKEM_Q-1].
   160	 *
   161	 * @spec{Normalizes on unsigned canonical representatives ahead of calling
   162	 * @[FIPS203, Compress_d, Eq (4.7)]. This is not made explicit in FIPS 203.}
   163	 *
   164	 * @param[in,out] r Input/output polynomial.
   165	 */
   166	/*
   167	 * NOTE: The semantics of mlk_poly_reduce() is different in
   168	 * the reference implementation, which requires
   169	 * signed canonical output data. Unsigned canonical
   170	 * outputs are better suited to the only remaining
   171	 * use of mlk_poly_reduce() in the context of (de)serialization.
   172	 */
   173	MLK_INTERNAL_API
   174	void mlk_poly_reduce(mlk_poly *r)
   175	__contract__(
   176	  requires(memory_no_alias(r, sizeof(mlk_poly)))
   177	  assigns(memory_slice(r, sizeof(mlk_poly)))
   178	  ensures(array_bound(r->coeffs, 0, MLKEM_N, 0, MLKEM_Q))
   179	);
